-- ========== ESP ==========
local espEnabled = false
local espDrawings = {}

local function clearESP()
    for _, d in pairs(espDrawings) do
        d.box:Remove(); d.name:Remove()
    end
    espDrawings = {}
end

local function createESP(plr)
    if plr == LocalPlayer or espDrawings[plr] then return end
    local box = Drawing.new("Square")
    box.Thickness = 1; box.Filled = false; box.Transparency = 1
    local name = Drawing.new("Text")
    name.Size = 13; name.Center = true; name.Outline = true; name.Transparency = 1
    espDrawings[plr] = {box = box, name = name}
end

local function updateESP()
    for plr, d in pairs(espDrawings) do
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not (hrp and hum and hum.Health > 0 and espEnabled) then
            d.box.Visible = false; d.name.Visible = false; continue
        end
        local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
        if not onScreen then
            d.box.Visible = false; d.name.Visible = false; continue
        end
        local head = char:FindFirstChild("Head")
        if not head then continue end
        local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local footPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
        local height = math.abs(headPos.Y - footPos.Y)
        local width = height * 0.6

        local isKiller = plr.Team and (plr.Team.Name:lower():find("killer") or plr.Team.Name:lower():find("murder"))
        local isJugg = plr.Team and plr.Team.Name:lower():find("jugg")
        local color = isJugg and Color3.fromRGB(200, 0, 255) or (isKiller and Color3.fromRGB(255, 50, 50) or Color3.fromRGB(50, 255, 100))

        d.box.Color = color
        d.box.Size = Vector2.new(width, height)
        d.box.Position = Vector2.new(pos.X - width/2, pos.Y - height/2)
        d.box.Visible = true
        d.name.Text = plr.Name
        d.name.Color = color
        d.name.Position = Vector2.new(pos.X, pos.Y - height/2 - 14)
        d.name.Visible = true
    end
end

-- ========== INVISIBILITY (toggleable, revert works) ==========
local invisEnabled = false
local invisRenderConn
local invisParts = {}

local function applyInvisibility(on)
    local char = LocalPlayer.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if on then
                if part.Transparency < 1 then
                    invisParts[part] = part.Transparency
                end
                part.Transparency = 1
            else
                part.Transparency = invisParts[part] or 0
            end
        end
        if part:IsA("Decal") or part:IsA("Texture") then
            if on then
                if part.Transparency < 1 then
                    invisParts[part] = part.Transparency
                end
                part.Transparency = 1
            else
                part.Transparency = invisParts[part] or 0
            end
        end
    end
    if not on then invisParts = {} end
end

local function setupInvisConn()
    if invisRenderConn then invisRenderConn:Disconnect() end
    invisRenderConn = RunService.RenderStepped:Connect(function()
        if invisEnabled then
            applyInvisibility(true)
        end
    end)
end

-- ========== GHOST MODE (extra functions for ghost tab) ==========
local ghostSpeed = false
local noClipEnabled = false
local speedValue = 22

local ghostSpeedConn
local function startGhostSpeed()
    if ghostSpeedConn then ghostSpeedConn:Disconnect() end
    ghostSpeedConn = RunService.Heartbeat:Connect(function()
        if not ghostSpeed then return end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.WalkSpeed = speedValue
        end
    end)
end

local noclipConn
local function setupNoclip()
    if noclipConn then noclipConn:Disconnect() end
    noclipConn = RunService.Stepped:Connect(function()
        if not noClipEnabled then return end
        local char = LocalPlayer.Character
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end)
end

-- ========== WEAPON LIST (collapsible within page) ==========
local weaponListExpanded = true
local weaponListContainer
local weaponScroll

local function getWeaponsOnMap()
    local weapons = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Tool") and obj.Parent and obj.Parent:IsA("Model") then
            local handle = obj:FindFirstChild("Handle")
            if handle then table.insert(weapons, obj) end
        end
    end
    return weapons
end

local function tryGrabWeapon(tool)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local handle = tool:FindFirstChild("Handle")
    if not (hrp and handle) then return end
    hrp.CFrame = CFrame.new(handle.Position + Vector3.new(0, 2, 0))
    task.wait(0.12)
    local prompt = tool:FindFirstChildOfClass("ProximityPrompt")
    if prompt then pcall(function() fireproximityprompt(prompt) end) end
    local cd = tool:FindFirstChildOfClass("ClickDetector")
    if cd then pcall(function() fireclickdetector(cd) end) end
    if handle and hrp then
        pcall(function() firetouchinterest(hrp, handle, 0) end)
        task.wait(0.05)
        pcall(function() firetouchinterest(hrp, handle, 1) end)
    end
end

local dragging, dragInput, dragStart, startPos
main.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
main.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
