    bar.BackgroundColor3 = Color3.fromRGB(30, 30, 44)

    bar.Parent = row
    Instance.new("UICorner", bar).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 240, 200)
    fill.BorderSizePixel = 0
    fill.Parent = bar
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function setValue(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * rel)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        lbl.Text = text .. ": " .. val
        callback(val)
    end
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setValue(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    return row
end

local function isMonsterName(name)
    local lowerName = name:lower()
    for _, monster in ipairs(MONSTER_LIST) do
        if lowerName:find(monster) then
            return true
        end
    end

-- ========== OPTIMIZED ESP ==========
local function clearESP()
    for _, d in pairs(espDrawings) do
        d.box:Remove(); d.name:Remove()
    end
    for _, h in pairs(highlights) do
        if h then h:Destroy() end
    end
    espDrawings = {}
    highlights = {}
end

local function createESP(plr)
    if plr == LocalPlayer or espDrawings[plr] then return end
    local box = Drawing.new("Square")
    box.Thickness = 1; box.Filled = false; box.Transparency = 1
    local name = Drawing.new("Text")
    name.Size = 13; name.Center = true; name.Outline = true; name.Transparency = 1
    espDrawings[plr] = {box = box, name = name}
    
    if plr.Character then
        local hl = Instance.new("Highlight")
        hl.Parent = plr.Character
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.Enabled = false
        highlights[plr] = hl
    end
end

local function updateESP()
    for plr, d in pairs(espDrawings) do
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        
        if char and not highlights[plr] then
            local hl = Instance.new("Highlight")
            hl.Parent = char
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.FillTransparency = 1
            hl.Enabled = false
            highlights[plr] = hl
        end
        
        if not (hrp and hum and hum.Health > 0 and espEnabled) then
            d.box.Visible = false; d.name.Visible = false
            if highlights[plr] then highlights[plr].Enabled = false end
            continue
        end
        
        local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
        if not onScreen then
            d.box.Visible = false; d.name.Visible = false
            if highlights[plr] then highlights[plr].Enabled = false end
            continue
        end
        
        local head = char:FindFirstChild("Head")
        if not head then continue end
        local hp = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
        local fp = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
        local h = math.abs(hp.Y - fp.Y)
        local w = h * 0.6
        
        local isKiller = false
        local isJugg = false
        
        -- 1. Проверка по команде
        if plr.Team then
            local tName = plr.Team.Name:lower()
            if tName:find("killer") or tName:find("murder") or tName:find("monster") or tName:find("demon") or tName:find("убий") then isKiller = true
            elseif tName:find("jugg") or tName:find("джаг") then isJugg = true end
        end

        -- 2. Проверка по списку монстров (Имя персонажа или DisplayName)
        local charName = char.Name:lower()
        local displayName = plr.DisplayName:lower()
        if isMonsterName(charName) or isMonsterName(displayName) then
            isKiller = true
        end

        -- 3. Проверка по оружию
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            local toolName = tool.Name:lower()
            if toolName:find("knife") or toolName:find("sword") or toolName:find("saw") or toolName:find("axe") or toolName:find("claw") or toolName:find("нож") or toolName:find("пила") then isKiller = true end
        end

        local color = Color3.fromRGB(50, 255, 100) -- Выживший
        if isJugg then color = Color3.fromRGB(200, 0, 255)
        elseif isKiller then color = Color3.fromRGB(255, 50, 50) end -- Монстры/Убийцы

        d.box.Color = color
        d.box.Size = Vector2.new(w, h)
        d.box.Position = Vector2.new(pos.X - w/2, pos.Y - h/2)
        d.box.Visible = true
        d.name.Text = plr.Name
        d.name.Color = color
        d.name.Position = Vector2.new(pos.X, pos.Y - h/2 - 14)
        d.name.Visible = true

        if highlights[plr] then
            highlights[plr].OutlineColor = color
            highlights[plr].Enabled = true
        end
    end
end

-- ========== INVISIBILITY ==========
local function applyInvisibility(on)
    local char = LocalPlayer.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if on then
                if part.Transparency < 1 then invisParts[part] = part.Transparency end
                part.Transparency = 1
            else
                part.Transparency = invisParts[part] or 0
            end
        end
        if part:IsA("Decal") or part:IsA("Texture") then
            if on then
                if part.Transparency < 1 then invisParts[part] = part.Transparency end
                part.Transparency = 1
            else
                part.Transparency = invisParts[part] or 0
            end
        end
    end
    if not on then invisParts = {} end
end

local function isPlayerTool(obj)
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character and obj:IsDescendantOf(plr.Character) then return true end
        local backpack = plr:FindFirstChildOfClass("Backpack")
        if backpack and obj:IsDescendantOf(backpack) then return true end
    end
    return false
end

local function isWeaponStand(obj)
    local name = obj.Name:lower()
    if name:find("ammo") or name:find("патрон") or name:find("patron") or name:find("door") or name:find("двер") or name:find("button") then return false end
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then
        local action = prompt.ActionText:lower()
        local objName = prompt.ObjectText:lower()
        if action:find("take") or action:find("grab") or action:find("взять") or action:find("подобр") then return true end
        if objName ~= "" and not objName:find("ammo") then return true end
    end
    if obj:IsA("Model") then
        if obj:FindFirstChildWhichIsA("Tool", true) then return true end
    end
    return false
end

local function getWeaponStands()
    local stands = {}
    local seen = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if isPlayerTool(obj) then continue end
        if isWeaponStand(obj) then
            local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
            local clickDet = obj:FindFirstChildWhichIsA("ClickDetector", true)
            if prompt or clickDet then
                local root = obj
                while root.Parent and root.Parent ~= Workspace and not root:IsA("Model") and not root:IsA("BasePart") do
                    root = root.Parent
                end
                local realWeaponName = "Weapon"
                if prompt and prompt.ObjectText ~= "" then realWeaponName = prompt.ObjectText else
                    local toolInside = root:FindFirstChildWhichIsA("Tool", true)
                    if toolInside then realWeaponName = toolInside.Name else realWeaponName = root.Name end
                end
                if realWeaponName:lower() == "take" or realWeaponName:lower() == "part" then realWeaponName = root.Name end
                if not seen[realWeaponName] and not realWeaponName:lower():find("ammo") then
                    seen[realWeaponName] = true
                    table.insert(stands, {root = root, display = realWeaponName, prompt = prompt, click = clickDet})
                end
            end
        end
    end
    return stands
end

-- ========== PHANTOM GRAB (Хитбокс улетает, камера стоит) ==========
local function grabFromStand(stand, isInstant)
    if isTeleporting then return end
    isTeleporting = true

    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then isTeleporting = false return end

    local targetPart = nil
    if stand.prompt then targetPart = stand.prompt.Parent
    elseif stand.click then targetPart = stand.click.Parent end
    if targetPart and not targetPart:IsA("BasePart") then targetPart = stand.root:FindFirstChildWhichIsA("BasePart", true) end
    if not targetPart then targetPart = stand.root:IsA("BasePart") and stand.root or stand.root:FindFirstChildWhichIsA("BasePart", true) end
    if not targetPart then isTeleporting = false return end

    local originalCFrame = hrp.CFrame
    local camCFrame = Camera.CFrame
    
    hrp.CFrame = CFrame.new(targetPart.Position)

    if stand.prompt then pcall(function() fireproximityprompt(stand.prompt) end) end
    if stand.click then pcall(function() fireclickdetector(stand.click) end) end

    hrp.CFrame = originalCFrame
    Camera.CFrame = camCFrame

    isTeleporting = false
end

local function executeAutoAmmo()
    if isTeleporting then return end
    local targetStand = nil
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if not isPlayerTool(obj) then
            local name = obj.Name:lower()
            local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
            local action = prompt and prompt.ActionText:lower() or ""
            local objText = prompt and prompt.ObjectText:lower() or ""
            if name:find("ammo") or name:find("патрон") or action:find("перезаряд") or objText:find("патрон") or objText:find("ammo") then
                targetStand = {root = obj, prompt = prompt, click = obj:FindFirstChildWhichIsA("ClickDetector", true)}
                break
            end
        end
    end
    if targetStand then grabFromStand(targetStand, true) end
end

-- ========== AUTO AMMO BUTTON ==========
local isAmmoBtnOut = false
local ammoBtnGui

local function createAmmoButton()
    if ammoBtnGui then ammoBtnGui:Destroy() end
    ammoBtnGui = Instance.new("ScreenGui")
    ammoBtnGui.Name = "SAKTK_AmmoBtn"
    ammoBtnGui.IgnoreGuiInset = true
    ammoBtnGui.ResetOnSpawn = false
    ammoBtnGui.Parent = CoreGui

    local ammoBtn = Instance.new("TextButton")
    ammoBtn.Size = UDim2.new(0, 50, 0, 50)
    ammoBtn.Position = UDim2.new(0, 20, 0, 200)
    ammoBtn.BackgroundColor3 = Color3.fromRGB(0, 180, 150)
    ammoBtn.Text = "🔋"
    ammoBtn.TextColor3 = Color3.new(1,1,1)
    ammoBtn.Font = Enum.Font.GothamBold
    ammoBtn.TextSize = 24
    ammoBtn.BorderSizePixel = 0
    ammoBtn.Parent = ammoBtnGui
    Instance.new("UICorner", ammoBtn).CornerRadius = UDim.new(1, 0)
    
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(255, 255, 255)
    stroke.Thickness = 2
    stroke.Transparency = 0.5
    stroke.Parent = ammoBtn

    local dragging, dragStart, startPos
    ammoBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = ammoBtn.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    ammoBtn.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            ammoBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    ammoBtn.MouseButton1Click:Connect(function() executeAutoAmmo() end)
end

local function toggleAmmoButton()
    isAmmoBtnOut = not isAmmoBtnOut
    if isAmmoBtnOut then createAmmoButton() else if ammoBtnGui then ammoBtnGui:Destroy() end end
end

-- ========== NEW FUNCTIONS ==========
local antiFlingEnabled = false
local autoInteractEnabled = false

RunService.Heartbeat:Connect(function()
    if antiFlingEnabled then
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            if hrp.Velocity.Magnitude > 100 or hrp.RotVelocity.Magnitude > 50 then
                hrp.Velocity = Vector3.new(0, 0, 0)
                hrp.RotVelocity = Vector3.new(0, 0, 0)
            end
        end
    end
    if autoInteractEnabled then
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("ProximityPrompt") and (obj.ObjectText:lower():find("health") or obj.ObjectText:lower():find("armor") or obj.ObjectText:lower():find("аптеч") or obj.ObjectText:lower():find("броня")) then
                    local dist = (obj.Parent.Position - hrp.Position).Magnitude
                    if dist < 10 then
                        pcall(function() fireproximityprompt(obj) end)
                    end
                end
            end
        end
    end
end)

local function toggleFullbright(on)
    if on then
        Lighting.Brightness = 3
        Lighting.ClockTime = 12
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 100000
        for _, v in ipairs(Lighting:GetChildren()) do
            if v:IsA("Atmosphere") then v.Density = 0 end
        end
    else
        Lighting.Brightness = 2
        Lighting.GlobalShadows = true
        Lighting.FogEnd = 1000
    end
end

local function clearFPS()
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("ParticleEmitter") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") or v:IsA("Decal") then
            v.Enabled = false
            if v:IsA("Decal") then v.Transparency = 1 end
        end
    end
    for _, v in ipairs(Lighting:GetDescendants()) do
        if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect") or v:IsA("SunRaysEffect") then
            v.Enabled = false
        end
    end
    pcall(function() settings().Rendering.QualityLevel = 1 end)
end

local function spamServer()
    for i = 1, 30 do
        local s = Instance.new("Sound")
        s.Parent = Workspace
        s.Volume = 10
        s.Playing = true
    end
end

    end
end)

-- ========== MAIN LOOP ==========
RunService.RenderStepped:Connect(function()
    if espEnabled then updateESP() end
    if invisEnabled then applyInvisibility(true) end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local tool = char and char:FindFirstChildOfClass("Tool")

    if noRecoilEnabled and tool then
        for _, d in ipairs(tool:GetDescendants()) do
            if d:IsA("NumberValue") and d.Name:lower():match("recoil") then d.Value = 0 end
        end
    end
    if rapidFireEnabled and tool then
        local fr = tool:FindFirstChild("FireRate") or tool:FindFirstChild("FireRateValue")
        if fr and (fr:IsA("NumberValue") or fr:IsA("IntValue")) then fr.Value = 0.01 end
    end
    if hitboxEnabled then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                if hrp then hrp.Size = Vector3.new(10, 10, 10); hrp.Transparency = 0.5; hrp.CanCollide = false end
            end
        end
    end
    if ghostSpeed and hum then hum.WalkSpeed = speedValue end
    if noClipEnabled and char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then part.CanCollide = false end
        end
    end
end)
